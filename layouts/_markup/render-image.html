{{- /* Image render hook. Photos live in assets/images/ so Hugo can process them. Originals are camera JPEGs (up to ~6 MB); this emits resized WebP variants plus a srcset so browsers fetch the smallest sufficient file, and lazy-loads anything below the fold. Images listed in site.Params.verbatim_images (the bank QR code) are published untouched so they stay pixel-exact, but still get width/height so they do not shift the layout. Anything not found in assets/ falls through to a plain unchanged. */ -}} {{- $src := .Destination -}} {{- $alt := .Text -}} {{- $title := .Title -}} {{- $res := resources.Get $src -}} {{- $verbatim := in (site.Params.verbatim_images | default slice) $src -}} {{- $processable := false -}} {{- with $res -}} {{- if and (not $verbatim) (in (slice "image/jpeg" "image/png") .MediaType.Type) -}} {{- $processable = true -}} {{- end -}} {{- end -}} {{- if $processable -}} {{- $widths := .Page.Param "content_image_widths" | default (slice 480 960 1440) -}} {{- $sizes := .Page.Param "content_image_sizes" | default "(min-width: 60em) 60em, 100vw" -}} {{- $srcset := slice -}} {{- range $widths -}} {{- if ge $res.Width . -}} {{- $v := $res.Resize (printf "%dx webp q82 Lanczos" .) -}} {{- $srcset = $srcset | append (printf "%s %dw" $v.RelPermalink $v.Width) -}} {{- end -}} {{- end -}} {{- /* Fallback src: capped at the largest variant for this display context. */ -}} {{- $fallback := $res -}} {{- $fallbackWidth := index $widths (sub (len $widths) 1) -}} {{- if gt $res.Width $fallbackWidth -}} {{- $fallback = $res.Resize (printf "%dx webp q82 Lanczos" $fallbackWidth) -}} {{- else -}} {{- $fallback = $res.Resize (printf "%dx webp q82" $res.Width) -}} {{- end -}} {{- if not $srcset -}} {{- $srcset = slice (printf "%s %dw" $fallback.RelPermalink $fallback.Width) -}} {{- end -}} {{- else if and $res $verbatim -}} {{- /* Published as-is; .Permalink on an unprocessed resource copies the file through untouched, so the bytes served are identical to the source. */ -}} {{- else -}} {{- end -}}
